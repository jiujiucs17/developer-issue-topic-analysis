import torch

import pycuda.autoinit
import pycuda.driver as cuda
from pycuda.compiler import sourcemodule

cuda_num_threads = 1024

def get_blocks(n):
    return (n + cuda_num_threads - 1) // cuda_num_threads

class holder(cuda.pointerholderbase):
    def __init__(self, t):
        super(holder, self).__init__()
        assert t.is_cuda
        self.t = t
        self.gpudata = t.data_ptr()
    def get_pointer(self):
        return self.t.data_ptr()

def complex_square(x):
    assert x.is_cuda
    assert x.size(-1) == 2
    batch_size = x.size()[:-1]

    x = x.view(-1, 2) # [batch, complex] (nbatch, 2)
    nbatch = x.size(0)

    cuda_kernel = _setup_complex_square_cuda_kernel(nbatch)

    output = torch.cuda.floattensor(nbatch, 2)
    cuda_kernel(holder(x), holder(output),
                block=(cuda_num_threads, 1, 1),
                grid=(get_blocks(nbatch), 1, 1))
    # [batch, complex] (nbatch, 2)

    output = output.view(*batch_size, 2)
    return output

@lru_cache(maxsize=32)
def _setup_complex_square_cuda_kernel(nbatch):
    kernel = '''
extern "c"
__global__ void main_(const float* in, float* out) {
    for (int index = blockidx.x * blockdim.x + threadidx.x; index < {nbatch}; index += blockdim.x * griddim.x) {
        float in_re = in[index * 2 + 0];
        float in_im = in[index * 2 + 1];
        out[index * 2 + 0] = in_re * in_re - in_im * in_im;
        out[index * 2 + 1] = 2.0 * in_re * in_im;
    }
}
'''.replace("{nbatch}", str(nbatch))
    
    return sourcemodule(kernel).get_function("main_")

    assert x.is_cuda
    assert x.size(-1) == 2
    batch_size = x.size()[:-1]

    x = x.view(-1, 2) # [batch, complex] (nbatch, 2)
    nbatch = x.size(0)

    cuda_kernel = _setup_complex_cube_cuda_kernel(nbatch)

    output = torch.cuda.floattensor(nbatch, 2)

context
a variety of improvements to distributeddataparallel, namely around ensuring local training parity and enhancing performance  where possible for certain cases are planned or already in the works. in particular, with respect to local training parity, we are aiming to support the following use cases:

not all outputs of ddp forward() used in loss computation. today, this results in an error during backward as these parameters are not marked as unused.
calling backward() twice. this is useful for a variety of use cases (i.e. computing different losses, loss with respect to different targets, etc) but ddp does not support this due to the way we account for autoggrad hooks.
activation checkpointing with multiple checkpoints, as well as activation checkpointing with unused parameters

for the above cases, it would be very useful to have a callback that fires when ddp user issues a call to backward, which helps ddp know that backwards is being called - currently, we only have access to the backwards pass via the autograd hooks installed on the reducer. such a callback will allow the following features:

traverse graph beginning from outputs to find complete set of unused parameters (unused parameters means parameters that don’t get gradient in this backward pass). this set will include the parameters that don’t get gradient because a particular model output was unused when computing loss.
calling backward() twice, by resetting the appropriate flags each time our callback fires. this mainly involves moving reducer::prepare_for_backward to be called by this function.
miscellaneous fixes such as moving backward start log time to this function.
enqueuing a delay_allreduce after backwards pass is over, which @zhaojuanmao is doing in enable static graph training in ddp #55248.

    def __torch_function__(
        self,

        types,
        args = (),
        kwargs = none,
    ):

class leakyrelufunction(function):
    @staticmethod
    def forward(ctx, input: th.tensor, negative_slope=1e-2) -> th.tensor:
        relu = functional.leaky_relu(input, negative_slope)
        ctx.save_for_backward(relu if negative_slope >= 0 else input)
        ctx.negative_slope = negative_slope
        return relu

    @staticmethod
    def backward(ctx, grad: th.tensor):
        relu, = ctx.saved_tensors
        negative_slope = ctx.negative_slope
        grad *= negative_slope * (relu <= 0) + (relu > 0)
        return grad, none


class leakyrelu(nn.module):

        super(leakyrelu, self).__init__()
        self.negative_slope = negative_slope

    def forward(self, input: th.tensor) -> th.tensor:
        return leakyrelufunction.apply(input, self.negative_slope)

🐛 describe the bug
@chillee nerdsniped me into thinking about how to get aotautograd working on higher order derivatives. for a moment, let's forget about the actual compilation aspect, and just focus on what it would take to trace out and then reconstruct a torch.autograd.function that supports higher order derivatives.
the basic hazard, as laid out in @zdevito's  https://colab.research.google.com/drive/1vpee6uveprz9hmshh1ks0xxxjyu533ec?usp=sharing is that saved tensors for backwards may receive gradient contributions when run in a higher order context. if we trace only a single backwards, we won't ever see such contributions (because they're always zero in this case.)
so what ingredients do we need for our torch.autograd.function?

we need to explicitly return every saved tensor for backwards as an output of the torch.autograd.function, so that they are able to receive gradients when used, and
we need a way to specify additional gradients for the saved tensors when we run backwards

if we have (1), we automatically have (2), since we can add the list of tensors to the outputs argument of torch.autograd.grad, and then just specify the extra gradient contributions we wanted to provide with grad_outputs. so all we need is something like torch.autograd.saved_for_backwards(outputs, inputs) which returns a list of all tensors which were saved for backwards given the graph specified by outputs and inputs, so that we can additionally return them as outputs.
then, aotautograd can simply perform this query, and add the saved tensors for backwards as extra outputs to the forward function they trace. graph partitioning is then trivial, as the cut points are exactly uses of these saved tensors.
